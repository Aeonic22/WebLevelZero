rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /neda_messages/{document=**} {
      allow read: if true;
      allow create: if exists(/databases/$(database)/documents/neda_allowedDevices/$(request.resource.data.deviceId));
      allow delete: if request.auth != null;
    }
    match /neda_allowedDevices/{document=**} {
      allow read: if false;
      allow write: if false;
    }
  }
}
